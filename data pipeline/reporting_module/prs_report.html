Prepare data

#only keep sampleset column != reference

df = df[df["sampleset"] != "reference"]
#remove _hmPOS_GRCh38 in the PGS column value strings

df["PGS"] = df["PGS"].str.replace("_hmPOS_GRCh38", "", regex=False)

print(df.columns)
print(meta_data.columns)
#join metadata on df by the PGS and Polygenic Score (PGS) ID column

df = df.merge(
    meta_data[["Polygenic Score (PGS) ID", "PGS Name", "Reported Trait", "PGS Publication (PGP) ID"]],
    left_on="PGS",
    right_on="Polygenic Score (PGS) ID",
    how="left"
)

print(df.columns)
print(publications_meta_data.columns)

#only keep the values in the "PGS Publication/Study (PGP) ID", whihc are present in the "Polygenic Score (PGS) ID" column of the df

    title="Percentile of Most Similar Population for Each PGS",
